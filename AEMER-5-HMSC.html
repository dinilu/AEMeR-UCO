Preparamos un poco los datos

comm <- as.matrix(com)

envm <- scale(env)
envm <- as.data.frame(envm)

Definimos el diseño experimental

library(Hmsc)
studyDesign <- data.frame(sitio = as.factor(sites$Sitio),
                          parcela = as.factor(sites$Parcela),
                          año = as.factor(sites$Año),
                         estacion = as.factor(sites$Estacion))
sitioL <- HmscRandomLevel(units = levels(sites$Sitio))
parcelaL <- HmscRandomLevel(units = levels(sites$Parcela))
añoL <- HmscRandomLevel(units = levels(sites$Año))
estacionL <- HmscRandomLevel(units = levels(sites$Estacion))


sitioL$nfMin <- 2
parcelaL$nfMin <- 2
añoL$nfMin <- 2
estacionL$nfMin <- 2

Especificamos la estructura del modelo

m0 <- Hmsc(Y = comm, 
         XData = envm, 
         XFormula = ~1,
         studyDesign = studyDesign, 
         ranLevels = list("sitio" = sitioL, 
                          "parcela" = parcelaL, 
                          "año" = añoL, 
                          "estacion" = estacionL),
         distr = "lognormal poisson")

m1 <- Hmsc(Y = comm, 
           XData = envm,
           XFormula = ~`N-NO3-` + `N-NH3` + `N-NO2-` +
             `PO43-` + pH + DO + EC + NTU + `T°C` +
             Current.velocity + Average.depth + Average.width,
           distr = "lognormal poisson")


m_env <- Hmsc(Y = comm, 
          XData = envm,
           XFormula = ~`N-NO3-` + `N-NH3` + `N-NO2-` +
            `PO43-` + pH + DO + EC + NTU + `T°C` +
             Current.velocity + Average.depth + Average.width,
          studyDesign = studyDesign, 
         ranLevels = list("sitio" = sitioL, 
                          "parcela" = parcelaL, 
                          "año" = añoL, 
                          "estacion" = estacionL),
           distr = "lognormal poisson")

Definimos los parámetros para las Cadenas de Markov Monte Carlo

samples <- 100 # Este valor debería ser más alto. Del orden de 10000 hacia arriba.
thin <- 1 # Este valor también suele ser más alto. Del orden de 50-100 hacia arriba.
transient <- 500*thin
nChains <- 4

Se calibran y estiman los parámetros del modelos con las MCMC

m0 <- sampleMcmc(m0, 
               thin = thin, 
               samples = samples, 
               transient = transient, 
               nChains = nChains, 
               nParallel = nChains)
m1 <- sampleMcmc(m1, 
               thin = thin, 
               samples = samples, 
               transient = transient, 
               nChains = nChains, 
               nParallel = nChains)
m_env <- sampleMcmc(m_env, 
               thin = thin, 
               samples = samples, 
               transient = transient, 
               nChains = nChains, 
               nParallel = nChains)

Convertimos los resultados en objetos CODA, son un tipo de datos de análisis bayesianos en R

mpost_m0 <- convertToCodaObject(m0)
mpost_m1 <- convertToCodaObject(m1)
mpost_m_env <- convertToCodaObject(m_env)

Calculamos dos estadísticos de calibración de MCMC

es_m0 <- effectiveSize(mpost_m0$Beta)
es_m1 <- effectiveSize(mpost_m1$Beta)
es_m_env <- effectiveSize(mpost_m_env$Beta)
gd_m0 <- gelman.diag(mpost_m0$Beta, multivariate=FALSE)$psrf
gd_m1 <- gelman.diag(mpost_m1$Beta, multivariate=FALSE)$psrf
gd_m_env <- gelman.diag(mpost_m_env$Beta, multivariate=FALSE)$psrf
